// --- 1. SWORD UN SHIELD (Tavs oriģinālais kods) ---
class Sword {
    int weight, quality;
public:
    Sword(int w = 2, int q = 1) : weight(w < 2 ? 2 : (w > 5 ? 5 : w)), quality(q < 1 ? 1 : (q > 6 ? 6 : q)) {}
    int Damage() { int d = weight * quality; if (quality > 1) quality--; return d; }
    void Sharpen() { quality = (quality + 2 > 6) ? 6 : quality + 2; }
    int GetQuality() { return quality; }
};

class Shield {
    int size, protection;
public:
    Shield(int s = 0, int p = 1) : size(s < 0 ? 0 : (s > 4 ? 4 : s)), protection(p < 1 ? 1 : (p > 4 ? 4 : p)) {}
    int Block() { int b = size * protection; if (size > 0) size--; return b; }
    void Repair() { size = (size + 2 > 4) ? 4 : size + 2; }
    int GetSize() { return size; }
};

// --- 2. CHARACTER KLASE AR OPERATORIEM ---
class Character {
public:
    // Noklusējuma un parametru konstruktors (1. attēla prasība)
    Character(string n = "Nezinams", int l = 100, int str = 10, int sw_w = 3, int sw_q = 3, int sh_s = 2, int sh_p = 2)
        : sw(sw_w, sw_q), sh(sh_s, sh_p), name(n), life(l < 0 ? 0 : l), strength(str), stepCount(0) {
        for (int i = 0; i < 10; i++) path[i] = '-';
        path[10] = '\0';
    }

    // Ievades/Izvades operatori (>> un <<)
    friend ostream& operator<<(ostream& os, const Character& ch) {
        os << ch.name << " (Life:" << ch.life << ", Str:" << ch.strength << ")";
        return os;
    }

    friend istream& operator>>(istream& is, Character& ch) {
        cout << "Ievadiet vardu un dzivibu: ";
        is >> ch.name >> ch.life;
        return is;
    }

    // Piešķiršanas operators (=)
    Character& operator=(const Character& otrs) {
        if (this != &otrs) {
            name = otrs.name; life = otrs.life; strength = otrs.strength;
        }
        return *this;
    }

    // Inkrements/Dekrements (++ un --)
    Character& operator++() { life++; return *this; } // ++ch
    Character operator++(int) { Character temp = *this; life++; return temp; } // ch++
    Character& operator--() { if(life > 0) life--; return *this; } // --ch
    Character operator--(int) { Character temp = *this; if(life > 0) life--; return temp; } // ch--

    // Salīdzināšanas operatori (pēc life * strength)
    bool operator<(const Character& otrs) const { return (life * strength) < (otrs.life * otrs.strength); }
    bool operator>(const Character& otrs) const { return (life * strength) > (otrs.life * otrs.strength); }
    bool operator<=(const Character& otrs) const { return !(*this > otrs); }
    bool operator>=(const Character& otrs) const { return !(*this < otrs); }
    bool operator==(const Character& otrs) const { return (life * strength) == (otrs.life * otrs.strength); }
    bool operator!=(const Character& otrs) const { return !(*this == otrs); }

    // Metodes
    bool Hit(int damage) { life = (life - damage < 0) ? 0 : life - damage; return life > 0; }
    int Attack() { return life > 0 ? strength + sw.Damage() : 0; }
    int Defence() { return life > 0 ? (strength / 2) + sh.Block() : 0; }
    int GetLife() const { return life; }
    void Go(char direction) { if (life > 0 && stepCount < 10) path[stepCount++] = direction; }
    
    virtual void Print() {
        cout << (life > 0 ? "[Dzivs] " : "[Miris] ") << name << " | HP: " << life << " | Speks: " << strength << endl;
    }
    virtual ~Character() {}

protected:
    string name;
    int life;
    int strength;
    char path[11];
    int stepCount;
    Sword sw;
    Shield sh;
};

// --- 3. MANTOŠANA (Monster, Hero, Boss) ---
class Monster : public Character {
    int level;
public:
    Monster(string n, int l, int str, int sw_w, int sw_q, int sh_s, int sh_p, int lvl)
        : Character(n, l, str, sw_w, sw_q, sh_s, sh_p), level(lvl) {}
    int GetLevel() { return level; }
    void Print() override { cout << "Monstrs: " << name << " [Lvl " << level << "] HP: " << life << endl; }
};

class Hero : public Character {
    int experience;
public:
    Hero(string n, int l, int str, int sw_w, int sw_q, int sh_s, int sh_p, int exp)
        : Character(n, l, str, sw_w, sw_q, sh_s, sh_p), experience(exp) {}
    void Kill(Monster &m) { experience += m.GetLife() * m.GetLevel(); m.Hit(m.GetLife()); }
    void Print() override { cout << "Varonis: " << name << " | HP: " << life << " | Exp: " << experience << endl; }
};

// --- 4. RINDAS (QUEUE) ŠABLONS (2. attēls) ---
template <typename Type>
class Queue {
private:
    Type* masiivs;
    int ietilpiiba;
    int skaits;
    int sakums;
    int beigas;
public:
    Queue(int izmers = 20) : ietilpiiba(izmers), skaits(0), sakums(0), beigas(-1) {
        masiivs = new Type[ietilpiiba];
    }
    ~Queue() { delete[] masiivs; }

    void enqueue(Type X) {
        if (skaits < ietilpiiba) {
            beigas = (beigas + 1) % ietilpiiba;
            masiivs[beigas] = X;
            skaits++;
        }
    }

    Type dequeue() {
        if (skaits > 0) {
            Type temp = masiivs[sakums];
            sakums = (sakums + 1) % ietilpiiba;
            skaits--;
            return temp;
        }
        return Type();
    }

    void PrintQueue() {
        for (int i = 0; i < skaits; i++) {
            cout << masiivs[(sakums + i) % ietilpiiba] << "   ";
        }
        cout << endl;
    }

    // Sortēšana (priekš 3. attēla uzdevuma)
    void Sort() {
        vector<Type> v;
        int n = skaits;
        for(int i=0; i<n; i++) v.push_back(dequeue());
        sort(v.begin(), v.end()); // Izmanto pārslogoto < operatoru
        for(auto &item : v) enqueue(item);
    }
};
